saveNoteBtn.addEventListener('click', () => {

const text = notesArea.value.trim();

if(!text) return alert('Write something sweet 💕');

const notes = JSON.parse(localStorage.getItem('love_notes') || '[]');

notes.unshift(text);

localStorage.setItem('love_notes', JSON.stringify(notes));

notesArea.value = '';

loadNotes();

});


clearNotesBtn.addEventListener('click', () => {

if(confirm('Clear all saved notes?')){

localStorage.removeItem('love_notes');

loadNotes();

}

});


savedNotesList.addEventListener('click', (e) => {

if(e.target.classList.contains('remove')){

const i = Number(e.target.dataset.i);

const notes = JSON.parse(localStorage.getItem('love_notes') || '[]');

notes.splice(i,1);

localStorage.setItem('love_notes', JSON.stringify(notes));

loadNotes();

}

});

// Gallery (store images in localStorage as data URLs)

function renderGallery(){

const imgs = JSON.parse(localStorage.getItem('love_images') || '[]');

thumbnails.innerHTML = '';

imgs.forEach(d => {

const img = document.createElement('img');

img.src = d;

thumbnails.appendChild(img);

});

}


imgUpload.addEventListener('change', (e) => {

const files = Array.from(e.target.files);

files.forEach(file => {

const reader = new FileReader();

reader.onload = function(evt){

const data = evt.target.result;

const imgs = JSON.parse(localStorage.getItem('love_images') || '[]');

imgs.unshift(data);

localStorage.setItem('love_images', JSON.stringify(imgs));

renderGallery();

}

reader.readAsDataURL(file);

});

});
